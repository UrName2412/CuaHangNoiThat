-- 5. Bảng CHUONGTRINHKHUYENMAI

    CCCD TEXT PRIMARY KEY,

-- 7. Bảng KHACHHANG
CREATE TABLE KHACHHANG (
    ID_khachHang SERIAL PRIMARY KEY,
    hoTen TEXT,
    SDT TEXT,
    email TEXT
);
ALTER TABLE KHACHHANG ENABLE ROW LEVEL SECURITY;


-- 8. Bảng DANGNHAP
CREATE TABLE DANGNHAP (
    tenDangNhap TEXT,
    loaiTaiKhoan TEXT CHECK (loaiTaiKhoan IN ('KHACHHANG', 'NHANVIEN')),
    matKhau TEXT NOT NULL,
    ID_khachHang INT REFERENCES KHACHHANG(ID_khachHang) ON DELETE CASCADE,
    CCCD_nhanVien TEXT REFERENCES NHANVIEN(CCCD) ON DELETE CASCADE,
    trangThai INT DEFAULT 1,
    ngayTao TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (tenDangNhap, loaiTaiKhoan),
    CONSTRAINT CHK_DANGNHAP_DOITUONG CHECK (
        (loaiTaiKhoan = 'KHACHHANG' AND ID_khachHang IS NOT NULL AND CCCD_nhanVien IS NULL) OR
        (loaiTaiKhoan = 'NHANVIEN' AND CCCD_nhanVien IS NOT NULL AND ID_khachHang IS NULL)
    )
);
ALTER TABLE DANGNHAP ENABLE ROW LEVEL SECURITY;


-- 9. Bảng DIACHIGIAOHANG (tinhThanh, phuongXa đổi sang INT để lưu mã ID từ API hành chính)
CREATE TABLE DIACHIGIAOHANG (
    ID_diaChi SERIAL PRIMARY KEY,
    ID_khachHang INT REFERENCES KHACHHANG(ID_khachHang) ON DELETE CASCADE,
    tinhThanh INT NOT NULL,   -- Lưu mã code Tỉnh/Thành phố từ API dropdown
    phuongXa INT NOT NULL,    -- Lưu mã code Quận/Huyện hoặc Phường/Xã từ API dropdown
    duong TEXT NOT NULL,      -- Lưu số nhà, tên đường do người dùng tự nhập
    macDinh INT DEFAULT 0 CHECK (macDinh IN (0, 1)) -- 1: Địa chỉ mặc định, 0: Địa chỉ phụ
);
ALTER TABLE DIACHIGIAOHANG ENABLE ROW LEVEL SECURITY;

-- ==========================================
-- 2. TẠO CÁC TRIGGERS TỰ ĐỘNG CẬP NHẬT isDeleted
-- ==========================================


CREATE OR REPLACE FUNCTION update_sanpham_is_deleted()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE SANPHAM
    SET isDeleted = 0
    WHERE ID_sanPham = NEW.ID_sanPham;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE TRIGGER trg_update_sanpham_hoadon
AFTER INSERT OR UPDATE ON CHITIETHOADON
FOR EACH ROW
EXECUTE FUNCTION update_sanpham_is_deleted();


CREATE OR REPLACE TRIGGER trg_update_sanpham_phieunhap
AFTER INSERT OR UPDATE ON CHITIETPHIEUNHAP
FOR EACH ROW
EXECUTE FUNCTION update_sanpham_is_deleted();




CREATE OR REPLACE FUNCTION update_nhacungcap_is_deleted()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE NHACUNGCAP
    SET isDeleted = 0
    WHERE ID_nhaCungCap = NEW.ID_nhaCungCap;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE TRIGGER trg_update_nhacungcap_phieunhap
AFTER INSERT OR UPDATE ON PHIEUNHAP
FOR EACH ROW
EXECUTE FUNCTION update_nhacungcap_is_deleted();




CREATE OR REPLACE FUNCTION update_danhmuc_tu_sanpham()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.ID_kieuDang IS NOT NULL THEN
        UPDATE KIEUDANG SET isDeleted = 0 WHERE ID_kieuDang = NEW.ID_kieuDang;
    END IF;
   
    IF NEW.ID_maGo IS NOT NULL THEN
        UPDATE LOAIGO SET isDeleted = 0 WHERE ID_maGo = NEW.ID_maGo;
    END IF;


    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE TRIGGER trg_update_danhmuc_tu_sanpham
AFTER INSERT OR UPDATE ON SANPHAM
FOR EACH ROW
EXECUTE FUNCTION update_danhmuc_tu_sanpham();




CREATE OR REPLACE FUNCTION update_loaisp_tu_kieudang()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.ID_loaiSP IS NOT NULL THEN
        UPDATE LOAISP SET isDeleted = 0 WHERE ID_loaiSP = NEW.ID_loaiSP;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE TRIGGER trg_update_loaisp_tu_kieudang
AFTER INSERT OR UPDATE ON KIEUDANG
FOR EACH ROW
EXECUTE FUNCTION update_loaisp_tu_kieudang();
